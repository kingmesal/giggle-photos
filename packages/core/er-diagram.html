    scanFolders {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        TEXT path UK "UNIQUE — absolute folder path"
        INTEGER priority "1-5, DEFAULT 3"
        TEXT status "added | scanning | ready | error"
        INTEGER scanSubdirectories "boolean 0/1, DEFAULT 1"
        INTEGER includeVideos "boolean 0/1, DEFAULT 1"
        INTEGER imageCount "DEFAULT 0"
        INTEGER videoCount "DEFAULT 0"
        INTEGER totalSize "bytes, DEFAULT 0"
        INTEGER createdAt "unix timestamp"
        INTEGER updatedAt "unix timestamp"
    }

    mediaFiles {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER scanFolderId FK "FK → scanFolders.id CASCADE"
        TEXT path UK "UNIQUE — absolute file path"
        TEXT filename "name without directory"
        TEXT extension "no dot (jpg, png, mp4)"
        TEXT fileType "image | video"
        INTEGER size "bytes"
        TEXT status "pending | metadata | hashed | processed | error"
        TEXT sha256Hash "nullable — exact file hash"
        TEXT contentHash "nullable — image content hash"
        TEXT perceptualHash "nullable — pHash"
        TEXT errorMessage "nullable"
        INTEGER takeoutFolderId FK "nullable — FK → takeoutFolders.id SET NULL"
        INTEGER createdAt "unix timestamp"
        INTEGER updatedAt "unix timestamp"
    }

    imageMetadata {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId FK,UK "FK → mediaFiles.id CASCADE, UNIQUE"
        INTEGER width "pixels"
        INTEGER height "pixels"
        TEXT format "jpeg, png, webp, gif …"
        TEXT colorSpace "nullable — sRGB, Adobe RGB …"
        INTEGER hasAlpha "boolean 0/1"
        INTEGER orientation "nullable — EXIF 1-8"
        INTEGER dateTaken "nullable — unix timestamp"
        TEXT cameraMake "nullable — EXIF"
        TEXT cameraModel "nullable — EXIF"
        REAL gpsLatitude "nullable"
        REAL gpsLongitude "nullable"
        REAL gpsAltitude "nullable — meters"
        INTEGER createdAt "unix timestamp"
        INTEGER updatedAt "unix timestamp"
    }

    videoMetadata {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId FK,UK "FK → mediaFiles.id CASCADE, UNIQUE"
        INTEGER width "pixels"
        INTEGER height "pixels"
        REAL duration "seconds"
        REAL frameRate "nullable — fps"
        TEXT codec "nullable — h264, hevc …"
        TEXT audioCodec "nullable — aac, mp3 …"
        INTEGER bitrate "nullable — bps"
        INTEGER dateTaken "nullable — unix timestamp"
        INTEGER createdAt "unix timestamp"
        INTEGER updatedAt "unix timestamp"
    }

    phashMatches {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId1 FK "FK → mediaFiles.id CASCADE"
        INTEGER mediaFileId2 FK "FK → mediaFiles.id CASCADE"
        INTEGER distance "hamming distance 0-64"
        INTEGER createdAt "unix timestamp"
    }

    phashGroupMembers {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId FK,UK "FK → mediaFiles.id CASCADE, UNIQUE"
        INTEGER groupAnchorFileId "lowest file ID in component (not FK)"
        INTEGER groupMinDistance "min hamming distance in group"
        INTEGER createdAt "unix timestamp"
    }

    fileDecisions {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId FK "FK → mediaFiles.id CASCADE"
        TEXT action "keep | delete"
        TEXT method "sha256 | content_hash | phash"
        TEXT reviewGroupKey "hash value or phash:anchorId"
        INTEGER createdAt "unix timestamp"
    }

    takeoutFolders {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER scanFolderId FK "FK → scanFolders.id CASCADE"
        TEXT rootPath "Google Photos directory"
        TEXT path UK "UNIQUE — folder path"
        TEXT kind "year | album | special | non_takeout"
        INTEGER year "nullable — year folders"
        TEXT albumTitle "nullable — metadata.json title or folder name"
        INTEGER createdAt "unix timestamp"
        INTEGER updatedAt "unix timestamp"
    }

    sidecars {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER scanFolderId FK "FK → scanFolders.id CASCADE"
        INTEGER takeoutFolderId FK "nullable — FK → takeoutFolders.id CASCADE"
        TEXT path UK "UNIQUE — JSON file path"
        TEXT kind "media | album_metadata | other | unparseable"
        TEXT title "nullable — original file name / album title"
        TEXT description "nullable"
        INTEGER photoTakenTime "nullable — unix timestamp"
        REAL geoLat "nullable"
        REAL geoLon "nullable"
        REAL geoAlt "nullable"
        INTEGER trashed "boolean 0/1"
        INTEGER archived "boolean 0/1"
        INTEGER favorited "boolean 0/1"
        TEXT people "JSON array of names"
        INTEGER mediaFileId FK "nullable — FK → mediaFiles.id SET NULL"
        TEXT matchMethod "nullable — name_exact | name_truncated | title | edited_name | album_reference"
        INTEGER createdAt "unix timestamp"
        INTEGER updatedAt "unix timestamp"
    }

    mediaPairs {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        TEXT kind "motion | edited"
        INTEGER primaryFileId FK "FK → mediaFiles.id CASCADE (still / original)"
        INTEGER secondaryFileId FK "FK → mediaFiles.id CASCADE (video / edited copy)"
        INTEGER createdAt "unix timestamp"
    }

    clipEmbeddings {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId FK "FK → mediaFiles.id CASCADE"
        INTEGER frameIndex "0 for images (video frames later)"
        TEXT modelId "model id incl. pinned revision"
        BLOB embedding "512 × float32, L2-normalised"
        INTEGER createdAt "unix timestamp"
    }

    clipMatches {
        INTEGER id PK "PRIMARY KEY AUTOINCREMENT"
        INTEGER mediaFileId1 FK "FK → mediaFiles.id CASCADE"
        INTEGER mediaFileId2 FK "FK → mediaFiles.id CASCADE"
        TEXT modelId "model id"
        REAL distance "cosine distance 0-2"
        INTEGER createdAt "unix timestamp"
    }
